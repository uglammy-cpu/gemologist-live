function LkTiles(container, opts){
  var self = this; opts = opts || {};
  this.box = container; this.tiles = {}; this.audios = document.createElement("div"); this.audios.className = "audios"; container.appendChild(this.audios);
  this.main = document.createElement("div"); this.main.className = "main"; container.appendChild(this.main);
  this.pips = document.createElement("div"); this.pips.className = "pips"; container.appendChild(this.pips);
  this.onChange = opts.onChange || function(){};
  // inserto = questa immagine sta nel riquadro piccolo anche se arriva per
  // prima: il gioiello del venditore tiene lo schermo grande, la gemmologa
  // che commenta sta nell'angolo. Senza questo vince chi si collega prima.
  this.add = function(id, el, label, mirror, inserto){
    if (self.tiles[id]) self.remove(id);
    if (mirror) el.className = (el.className || "") + " mirror";
    var wrap = document.createElement("div"); wrap.appendChild(el); wrap.title = label || "";
    wrap.setAttribute("data-inserto", inserto ? "1" : "0");
    self.tiles[id] = wrap;
    var cur = self.main.firstChild;
    // Chi è solo in stanza prende lo schermo grande, anche la gemmologa:
    // un riquadro grande vuoto con un francobollo nell'angolo non si guarda.
    if (!cur || (!inserto && cur.getAttribute("data-inserto") === "1")){
      if (cur){ cur.className = "pip"; self.pips.insertBefore(cur, self.pips.firstChild); }
      wrap.className = ""; self.main.appendChild(wrap);
    } else {
      wrap.className = "pip"; self.pips.appendChild(wrap);
    }
    wrap.addEventListener("click", function(){ self.promote(id); });
    self.onChange(self.count());
  };
  this.promote = function(id){
    var w = self.tiles[id]; if (!w || w.parentNode === self.main) return;
    var cur = self.main.firstChild; self.main.appendChild(w); w.className = "";
    if (cur){ cur.className = "pip"; self.pips.insertBefore(cur, self.pips.firstChild); }
  };
  this.remove = function(id){
    var w = self.tiles[id]; if (!w) return; self.tiles[id] = null;
    var wasMain = w.parentNode === self.main; w.remove();
    if (wasMain && self.pips.firstChild){ var n = self.pips.firstChild; self.main.appendChild(n); n.className = ""; }
    self.onChange(self.count());
  };
  this.addAudio = function(id, el){ el.setAttribute("data-id", id); self.audios.appendChild(el); };
  this.removeAudio = function(id){ var a = self.audios.querySelector('[data-id="' + id + '"]'); if (a) a.remove(); };
  this.count = function(){ return Object.keys(self.tiles).filter(function(k){ return !!self.tiles[k]; }).length; };
  this.mainVideo = function(){ return self.main.querySelector("video"); };
  this.clear = function(){ Object.keys(self.tiles).forEach(self.remove); self.audios.innerHTML = ""; };
}
function lkAttachRemote(room, tiles, LK){
  function onTrack(track, pub, participant){
    var id = participant.identity + ":" + (pub.trackSid || track.sid);
    if (track.kind === "video"){ var el = track.attach(); el.muted = true; el.setAttribute("playsinline", ""); tiles.add(id, el, participant.name || participant.identity, false, participant.identity.indexOf("staff:gemologist:") === 0); }
    else if (track.kind === "audio"){ tiles.addAudio(id, track.attach()); }
  }
  function offTrack(track, pub, participant){
    var id = participant.identity + ":" + (pub.trackSid || track.sid);
    track.detach().forEach(function(el){ el.remove(); });
    if (track.kind === "video") tiles.remove(id); else tiles.removeAudio(id);
  }
  room.on(LK.RoomEvent.TrackSubscribed, onTrack);
  room.on(LK.RoomEvent.TrackUnsubscribed, offTrack);
  room.on(LK.RoomEvent.ParticipantDisconnected, function(p){ Object.keys(tiles.tiles).forEach(function(id){ if (id.indexOf(p.identity + ":") === 0) tiles.remove(id); }); });
  room.on(LK.RoomEvent.Disconnected, function(){ tiles.clear(); });
}
